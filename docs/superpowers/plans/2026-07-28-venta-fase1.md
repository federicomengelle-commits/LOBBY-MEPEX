**Spec:** `docs/circuito-venta-blueprint.md` — este plan implementa la Fase 1 (§8).

**Objetivo:** que la venta exista como entidad visible, con su ficha, naciendo de
un paso de confirmación explícito en el CRM.

**Arquitectura:** tabla `ventas` como cabecera que apunta (no copia) cotización,
cliente, evento y proyectos. Módulo `#ventas` con listado (`ventas.js`) y ficha
(`venta-detalle.js`, con `renderInto` para servir también de modal). El
`plan_cobro` pasa a colgar de la venta con fallback a proyecto. **Cero
contabilidad en esta fase.**

**Stack:** vanilla JS ES6+, objetos globales, template literals, Supabase JS SDK
v2. Sin build, sin framework, sin tests unitarios.

## Restricciones globales

Copiadas del blueprint y del CLAUDE.md. **Aplican a todas las tareas.**

1. **Todo nace inerte** (§9.1). Columnas nullable. Si nadie usa lo nuevo, el
   sistema se comporta igual que hoy.

5. **Módulo nuevo necesita grant** en `roles.permissions`. Sin grant es invisible
   — mismo gotcha que `rendimiento` y `calendario-adm`.
6. **Escapar todo dato de usuario** en innerHTML con `escHtml` / `escAttr`
   (globales de `components.js`).
7. **Reviewers antes de commitear** (regla 19 CLAUDE.md §8): `sql-reviewer` para

   `security-reviewer` si toca input/auth/datos sensibles. CRITICAL/HIGH se
   arreglan antes del push.
8. **Ninguna view nueva sin `security_invoker = true`** (regla de facto desde
   2026-07-26 — las views bypassean RLS por default y eso ya causó una fuga real).
9. **`Auth.getUser().uid`** es el UUID de auth; **`.id`** es el username (`"fede"`).
   Para FKs a `profiles`/`auth` usar `.uid`.
10. **Ciclo de verificación** (no hay tests): `node --check <archivo>` → preview con

CREATE TABLE IF NOT EXISTS venta_numerador (
    anio   INT PRIMARY KEY,
    ultimo INT NOT NULL DEFAULT 0

CREATE OR REPLACE FUNCTION siguiente_numero_venta()
RETURNS TEXT
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
    v_anio INT := EXTRACT(YEAR FROM CURRENT_DATE)::INT;
    v_num  INT;
BEGIN
    INSERT INTO venta_numerador (anio, ultimo)
    VALUES (v_anio, 1)
    ON CONFLICT (anio) DO UPDATE SET ultimo = venta_numerador.ultimo + 1
    RETURNING ultimo INTO v_num;
    RETURN 'VTA-' || v_anio || '-' || LPAD(v_num::TEXT, 4, '0');

        }
        items.forEach(i => {
            i.canal = i.comprobante_venta_id
                ? (canalPorComp[i.comprobante_venta_id] || venta.canal_sugerido)
                : venta.canal_sugerido;
        });

        const total     = Number(venta.total) || 0;
        const facturado = items.filter(i => i.comprobante_venta_id)
                               .reduce((a, i) => a + (Number(i.monto) || 0), 0);
        const cobrado   = items.reduce((a, i) => a + (Number(i.monto_cobrado) || 0), 0);

        const canales = new Set(items.map(i => i.canal));
        const canal = canales.size === 0 ? venta.canal_sugerido
                    : canales.size > 1   ? 'mixta'
                    : [...canales][0];

        const primera = items[0] || null;
        const senaCobrada = !!(primera && primera.estado === 'cobrado');

        return { total, facturado, cobrado, saldo: total - cobrado,
                 canal, senaCobrada, cuotas: items };

    async confirmarVenta(id) {

        if (v.estado !== 'borrador') return { error: 'La venta ya está confirmada' };
        if (!v.cliente_id) return { error: 'Falta el cliente' };
        if (!(Number(v.total) > 0)) return { error: 'El total tiene que ser mayor a cero' };

        let numero = v.numero;
        if (!numero) {
            const { data, error } = await supabaseClient.rpc('siguiente_numero_venta');
            if (error) { console.warn('[API] siguiente_numero_venta:', error.message); throw error; }
            numero = data;
        }

        AuditLog.record('update', 'ventas', `Venta ${numero} confirmada`, 'venta', id);

        if (!motivo || !motivo.trim()) return { error: 'Hace falta el motivo' };

        if (!v) return { error: 'Venta no encontrada' };
        if (v.estado === 'anulada') return { error: 'La venta ya está anulada' };

        const r = await this.getVentaResumen(id);
        const facturasVivas = (r.cuotas || []).filter(c => c.comprobante_venta_id);
        if (facturasVivas.length) {
            const { data: comps } = await supabaseClient.from('comprobantes')
                .select('id, numero, canal, estado')
                .in('id', facturasVivas.map(c => c.comprobante_venta_id))
                .eq('canal', 'oficial').eq('estado', 'emitida').eq('_deleted', false);
            if ((comps || []).length) {
                return { error: 'Hay ' + comps.length + ' factura(s) oficial(es) emitida(s). ' +
                                'Emitir la nota de crédito antes de anular.',
                         comprobantes: comps };
            }
        }
        await this.updateVenta(id, { estado: 'anulada', motivo_anulacion: motivo.trim() });
        AuditLog.record('update', 'ventas', `Venta ${v.numero} anulada: ${motivo}`, 'venta', id);
        return { ok: true };
    },

    async deleteVenta(id, label) {
        const r = await this.getVentaResumen(id);
        if ((r.cuotas || []).some(c => c.comprobante_venta_id)) {
            return { error: 'No se puede borrar una venta con comprobantes. Anulala.' };
        }
        await UndoHelpers.deleteRecord('ventas', id, label || 'Venta');
        return { ok: true };
